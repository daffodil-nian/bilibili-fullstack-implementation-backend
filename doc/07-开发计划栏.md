 public void channelRead(final ChannelHandlerContext ctx, Object msg) throws Exception {
        HttpObject httpObject = (HttpObject)msg;
        if (httpObject instanceof HttpRequest) {
            final HttpRequest req = (HttpRequest)httpObject;
            this.isWebSocketPath = this.isWebSocketPath(req);
            if (!this.isWebSocketPath) {
                ctx.fireChannelRead(msg);
                return;
            }

            try {
                WebSocketServerHandshakerFactory wsFactory = new WebSocketServerHandshakerFactory(getWebSocketLocation(ctx.pipeline(), req, this.serverConfig.websocketPath()), this.serverConfig.subprotocols(), this.serverConfig.decoderConfig());
                final WebSocketServerHandshaker handshaker = wsFactory.newHandshaker(req);
                final ChannelPromise localHandshakePromise = this.handshakePromise;
                if (handshaker == null) {
                    WebSocketServerHandshakerFactory.sendUnsupportedVersionResponse(ctx.channel());
                } else {
                    WebSocketServerProtocolHandler.setHandshaker(ctx.channel(), handshaker);
                    ctx.pipeline().remove(this);
                    ChannelFuture handshakeFuture = handshaker.handshake(ctx.channel(), req);
                    handshakeFuture.addListener(new ChannelFutureListener() {
                        public void operationComplete(ChannelFuture future) {
                            if (!future.isSuccess()) {
                                localHandshakePromise.tryFailure(future.cause());
                                ctx.fireExceptionCaught(future.cause());
                            } else {
                                localHandshakePromise.trySuccess();
                                ctx.fireUserEventTriggered(ServerHandshakeStateEvent.HANDSHAKE_COMPLETE);
                                ctx.fireUserEventTriggered(new WebSocketServerProtocolHandler.HandshakeComplete(req.uri(), req.headers(), handshaker.selectedSubprotocol()));
                            }

                        }
                    });
                    this.applyHandshakeTimeout();
                }
            } finally {
                ReferenceCountUtil.release(req);
            }
        } else if (!this.isWebSocketPath) {
            ctx.fireChannelRead(msg);
        } else {
            ReferenceCountUtil.release(msg);
        }

    }

public class HttpHeadersHandler extends ChannelInboundHandlerAdapter {

    @Override
    public void channelRead(ChannelHandlerContext ctx, Object msg) throws Exception {
        if (msg instanceof FullHttpRequest request) {
            UrlBuilder urlBuilder = UrlBuilder.ofHttp(request.uri());

            String token = Optional.ofNullable(urlBuilder.getQuery())
                    .map(q -> q.get("token"))
                    .map(CharSequence::toString)
                    .orElse("");
            NettyUtil.setAttr(ctx.channel(), NettyUtil.TOKEN, token);

            // 去掉 query，只留 /ws 给协议升级
            request.setUri(urlBuilder.getPath().toString());

            String ip = request.headers().get("X-Real-IP");
            if (StrUtil.isBlank(ip)) {
                InetSocketAddress addr = (InetSocketAddress) ctx.channel().remoteAddress();
                ip = addr.getAddress().getHostAddress();
            }
            NettyUtil.setAttr(ctx.channel(), NettyUtil.IP, ip);

            ctx.pipeline().remove(this);
            ctx.fireChannelRead(request);
        } else {
            ctx.fireChannelRead(msg);
        }
    }
}

【可以这么理解，首先很多人以为IM单聊必须使用消息队列，然后说结论：“不一定”。接下来就要解释原理，并且多种方案对比，总结完优缺点最后一句话小结，6个步骤是技术博客的一个写法套路】

> 很多同学在实现 IM 单聊的时候，会认为单聊必须引入消息队列。 实际上单聊并不是一定要使用 MQ，Redis Pub/Sub 也可以实现基础的消息发布订阅，二者选型取决于业务规模与可靠性要求。
>
> 为什么不少 IM 系统会选择消息队列？ 消息队列是典型生产者消费者模型，能够实现消息发送方与接收方的解耦。同时 MQ 原生提供消息持久化、消息 ACK、消息堆积与重试能力。消息发送之后可以持久保存，用户离线，上线后可以重新消费消息。
>
> 而 Redis Pub/Sub 虽然同样是发布订阅模式，能够实现在线消息推送，但它的消息不会**持久化**。一旦客户端断开订阅，期间的消息会直接丢失。如果要支持离线消息，需要我们额外使用 Redis List/ZSet 来存储离线消息，并且自己实现消息重试、消息去重等逻辑，增加业务代码复杂度。
>
> 简单总结： 不是 MQ 的推拉操作本身更简单，而是 MQ 自带大量消息可靠性能力，可以减少自研成本；Redis PubSub 适合小型演示项目，高并发、要求消息不丢失的 IM 场景，一般会考虑 MQ。

| 随机推荐无感 | 刷新结果要有变化；可加权 |
| 又想做表情花活 | 拒绝；主路径已够答辩演示 |
